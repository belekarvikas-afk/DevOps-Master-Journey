# `01-provider/main.tf`

terraform {

  required_providers {

    azurerm = {
      source  = "hashicorp/azurerm"
      version = "~>4.0"
    }

  }

}

provider "azurerm" {

  features {}

  subscription_id = "YOUR_SUBSCRIPTION_ID"

}
```
